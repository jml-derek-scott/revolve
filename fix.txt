```
# In pipeline/texture_bake.py
def project_vertex_to_photo(p_world, view, mannequin_bounds):
    mlo, mhi = mannequin_bounds
    full_height = mhi[1] - mlo[1]
    
    # Scale based on the FULL person height in the photo
    scale = (view["bottom"] - view["top"]) / full_height
    
    # Map relative to floor and torso center
    px = view["cx"] + (p_world[0] - (mlo[0] + mhi[0]) / 2) * scale
    py = view["bottom"] - (p_world[1] - mlo[1]) * scale
    return int(round(px)), int(round(py))
	
```
